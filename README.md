  <script>
    const $ = id => document.getElementById(id);

    const weatherInfo = code => {
      if (code === 0) return ["☀️", "Clear"];
      if (code === 1) return ["🌤️", "Mainly clear"];
      if (code === 2) return ["⛅", "Partly cloudy"];
      if (code === 3) return ["☁️", "Cloudy"];
      if ([45, 48].includes(code)) return ["🌫️", "Foggy"];
      if ([51, 53, 55, 56, 57].includes(code))
        return ["🌦️", "Drizzle"];
      if ([61, 63, 65, 66, 67].includes(code))
        return ["🌧️", "Rain"];
      if ([71, 73, 75, 77].includes(code))
        return ["🌨️", "Snow"];
      if ([80, 81, 82].includes(code))
        return ["🌦️", "Rain showers"];
      if ([85, 86].includes(code))
        return ["🌨️", "Snow showers"];
      if ([95, 96, 99].includes(code))
        return ["⛈️", "Thunderstorm"];
      return ["🌡️", "Unknown"];
    };

    function addCell(row, value, isHeader = false) {
      const cell = document.createElement(isHeader ? "th" : "td");
      cell.textContent = value;
      row.appendChild(cell);
      return cell;
    }

    function resetRows() {
      $("hoursRow").replaceChildren();
      $("weatherRow").replaceChildren();
      $("conditionRow").replaceChildren();
      $("temperatureRow").replaceChildren();
      $("windRow").replaceChildren();

      addCell($("hoursRow"), "Hour", true);
      addCell($("weatherRow"), "Weather");
      addCell($("conditionRow"), "Condition");
      addCell($("temperatureRow"), "Temperature");
      addCell($("windRow"), "Wind");
    }

    async function loadWeather() {
      const city = $("cityInput").value.trim();

      if (!city) {
        $("message").textContent = "Please enter a city name.";
        $("message").className = "error";
        return;
      }

      $("message").textContent = "Loading weather data...";
      $("message").className = "loading";
      $("currentSummary").textContent = "Loading...";
      $("searchBtn").disabled = true;

      try {
        const geoURL = new URL(
          "https://geocoding-api.open-meteo.com/v1/search"
        );
        geoURL.search = new URLSearchParams({
          name: city,
          count: "1",
          language: "en",
          format: "json"
        });

        const geoResponse = await fetch(geoURL);
        if (!geoResponse.ok) throw new Error("Geocoding unavailable");

        const geoData = await geoResponse.json();
        const place = geoData.results?.[0];

        if (!place) {
          throw new Error("City not found. Try another city name.");
        }

        const params = new URLSearchParams({
          latitude: place.latitude,
          longitude: place.longitude,
          hourly: "temperature_2m,weather_code,wind_speed_10m",
          current: "temperature_2m,weather_code,wind_speed_10m",
          timezone: "auto",
          forecast_days: "2"
        });

        const response = await fetch(
          "https://api.open-meteo.com/v1/forecast?" + params
        );

        if (!response.ok) throw new Error("Weather data unavailable");

        const data = await response.json();
        const hourly = data.hourly;
        const now = data.current;

        const [icon, description] = weatherInfo(now.weather_code);

        $("currentIcon").textContent = icon;
        $("currentSummary").textContent =
          `${description} · ${now.temperature_2m} °C`;

        $("locationName").textContent =
          `${place.name}, ${place.country}`;

        const localNow = new Date(
          new Date(now.time).getTime()
        );

        $("date").textContent =
          `${place.name}, ${place.country} — ` +
          localNow.toLocaleDateString("vi-VN", {
            timeZone: "UTC",
            day: "2-digit",
            month: "2-digit",
            year: "numeric"
          });

        resetRows();

        // Hiển thị 24 giờ, bắt đầu từ giờ gần nhất.
        let start = hourly.time.findIndex(
          time => time.slice(0, 13) === now.time.slice(0, 13)
        );

        if (start < 0) {
          start = hourly.time.findIndex(time => time >= now.time);
        }
        if (start < 0) start = 0;

        for (let i = start; i < Math.min(start + 24, hourly.time.length); i++) {
          const time = hourly.time[i];
          const [hourIcon, condition] = weatherInfo(
            hourly.weather_code[i]
          );

          addCell(
            $("hoursRow"),
            time.slice(11, 16),
            true
          );

          const iconCell = addCell($("weatherRow"), hourIcon);
          iconCell.className = "weather-icon";

          addCell($("conditionRow"), condition);

          addCell(
            $("temperatureRow"),
            `${hourly.temperature_2m[i]} °C`
          );

          addCell(
            $("windRow"),
            `${hourly.wind_speed_10m[i]} km/h`
          );
        }

        $("message").textContent = "";
        $("message").className = "";
      } catch (error) {
        $("message").textContent =
          error.message || "Unable to load weather data.";
        $("message").className = "error";
        $("currentSummary").textContent =
          "Could not load the forecast.";
      } finally {
        $("searchBtn").disabled = false;
      }
    }

    $("searchBtn").addEventListener("click", loadWeather);

    $("cityInput").addEventListener("keydown", event => {
      if (event.key === "Enter") loadWeather();
    });

    loadWeather();
  </script>
